cmake_minimum_required(VERSION 3.22.1)
project(vulkan_arena_api LANGUAGES CXX)
add_library(vulkan_arena_api STATIC src/policy.cpp src/arena_policy.cpp src/arena_texture.cpp src/arena_pipeline.cpp src/arena_diagnostics.cpp )
set_target_properties(vulkan_arena_api PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(vulkan_arena_api PUBLIC cxx_std_17)
target_include_directories(vulkan_arena_api PUBLIC include)
target_compile_options(vulkan_arena_api PRIVATE -Wall -Wextra -Wpedantic)
if(ANDROID)
  target_link_libraries(vulkan_arena_api PUBLIC vulkan)
else()
  find_package(Vulkan REQUIRED)
  target_link_libraries(vulkan_arena_api PUBLIC Vulkan::Vulkan)
endif()
# Optional copy API is opt-in at link time; the game does not compile/link it.
add_library(vulkan_arena_compute STATIC EXCLUDE_FROM_ALL src/arena_compute.cpp)
set_target_properties(vulkan_arena_compute PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(vulkan_arena_compute PUBLIC vulkan_arena_api)
option(ARENA_API_TESTS "Build CPU contract tests" OFF)
if(ARENA_API_TESTS)
  enable_testing()
  add_executable(arena_api_test tests/api_test.cpp)
  target_link_libraries(arena_api_test PRIVATE vulkan_arena_api)
  add_test(NAME arena_api_test COMMAND arena_api_test)
endif()
