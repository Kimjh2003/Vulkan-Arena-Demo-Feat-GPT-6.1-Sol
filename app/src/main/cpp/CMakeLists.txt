cmake_minimum_required(VERSION 3.22.1)
project(arena)
# Shared data contract is independent from gameplay and Android/Vulkan ownership.
add_library(arena_scene_contract INTERFACE)
target_include_directories(arena_scene_contract INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/vulkan/include)
target_compile_features(arena_scene_contract INTERFACE cxx_std_17)
add_library(arena_gameplay INTERFACE)
target_include_directories(arena_gameplay INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(arena_gameplay INTERFACE cxx_std_17)

add_subdirectory(vulkan)
add_library(arena_graphics STATIC vulkan_renderer.cpp vulkan_context.cpp vulkan_swapchain.cpp vulkan_floor.cpp)
set_target_properties(arena_graphics PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(arena_graphics PUBLIC arena_scene_contract vulkan_arena_api PRIVATE android log)

add_library(arena_runtime STATIC game_runtime.cpp)
set_target_properties(arena_runtime PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(arena_runtime PUBLIC arena_gameplay PRIVATE arena_graphics)
add_library(arena SHARED native.cpp)
target_link_libraries(arena PRIVATE arena_runtime)

include(CheckIPOSupported)
check_ipo_supported(RESULT arena_ipo_supported OUTPUT arena_ipo_error)
if(arena_ipo_supported)
    set_property(TARGET vulkan_arena_api vulkan_arena_compute arena_graphics arena_runtime arena PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
    set_property(TARGET vulkan_arena_api vulkan_arena_compute arena_graphics arena_runtime arena PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELWITHDEBINFO TRUE)
endif()
foreach(arena_target IN ITEMS vulkan_arena_api vulkan_arena_compute arena_graphics arena_runtime arena)
    target_compile_options(${arena_target} PRIVATE -fvisibility=hidden -fvisibility-inlines-hidden
        -ffunction-sections -fdata-sections)
endforeach()
set(arena_exports "$<IF:$<CONFIG:Debug>,${CMAKE_CURRENT_SOURCE_DIR}/arena.debug.exports.map,${CMAKE_CURRENT_SOURCE_DIR}/arena.exports.map>")
target_link_options(arena PRIVATE -Wl,--gc-sections
    "-Wl,--version-script=${arena_exports}" -Wl,--no-undefined-version)
set_property(TARGET arena APPEND PROPERTY LINK_DEPENDS "${arena_exports}")
